async function getScoutingPlayers(): Promise<ScoutingPlayer[]> {
  const settings = await listRecords('app_settings');
  const saved = settings.find(item => item.key === SCOUTING_SETTING_KEY);
  if (saved && Array.isArray(saved.value)) return saved.value as ScoutingPlayer[];
  const initial = DEFAULT_SCOUTING_PLAYERS.map(player => ({ ...player }));
  await putRecord('app_settings', { ...base(), key: SCOUTING_SETTING_KEY, value: initial });
  return initial;
}
async function saveScoutingPlayers(players: ScoutingPlayer[]): Promise<void> {
  const settings = await listRecords('app_settings');
  const saved = settings.find(item => item.key === SCOUTING_SETTING_KEY);
  await putRecord('app_settings', saved
    ? { ...saved, updatedAt: nowISO(), value: players }
    : { ...base(), key: SCOUTING_SETTING_KEY, value: players });
}
function formatScoutingValue(value: number | null): string {
  return value === null || !Number.isFinite(value) ? 'ยังไม่ระบุ' : '€' + value.toLocaleString('en-US', { maximumFractionDigits: 1 }) + 'M';
}

function scoutingPhoto(player: ScoutingPlayer, size: 'card'|'profile' = 'card'): string {
  const cls = size === 'card' ? 'scout-photo scout-photo-card' : 'scout-photo scout-photo-profile';
  return `<div class="${cls}"><img src="${esc(player.photoUrl)}" alt="${esc(player.name)}" loading="lazy" referrerpolicy="no-referrer" onerror="this.style.display='none';this.parentElement.classList.add('photo-failed')"><span>${esc(player.name.split(' ').map(part => part[0]).join('').slice(0,2))}</span></div>`;
}
