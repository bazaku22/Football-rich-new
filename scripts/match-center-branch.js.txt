                  tx.oncomplete = () => resolve();
                  tx.onerror = () => reject(processingError || tx.error || new Error('บันทึกผลไม่สำเร็จ'));
                  tx.onabort = () => reject(processingError || tx.error || new Error('ยกเลิกการบันทึกผล'));
                });
                const matchReq = tx.objectStore('matches').getAll();
                const resultReq = tx.objectStore('match_results').getAll();
                const predictionReq = tx.objectStore('predictions').getAll();
                const performanceReq = tx.objectStore('performance_records').getAll();
                const requests = [matchReq,resultReq,predictionReq,performanceReq];
                let ready = 0;
                const process = () => {
                  try {
                    const currentMatches = matchReq.result;
                    const currentResults = resultReq.result;
                    const currentPredictions = predictionReq.result;
                    const currentPerformance = performanceReq.result;
                    if (currentResults.some(r => r.importResultId === payload.result_id)) throw new Error('Result ID นี้ถูกนำเข้าไปแล้ว');
                    const correctionAllowed = !!document.querySelector('#allow-result-correction')?.checked;
                    for (const entry of resolved) {
                      const match = currentMatches.find(m => m.id === entry.match.id);
                      if (!match) throw new Error('คู่แข่งขันถูกลบหรือเปลี่ยนแปลงระหว่างตรวจสอบ');
                      const item = entry.item;
                      const old = currentResults.find(r => r.matchId === match.id);
                      if (old && !correctionAllowed) throw new Error('มีผลเดิมอยู่แล้ว ต้องยืนยันการแก้ไขก่อน');
                      const history = Array.isArray(old?.revisionHistory) ? old.revisionHistory.slice() : [];
                      if (old) history.push({revision:Number(old.revisionNumber || 1), recordedAt:nowISO(), resultId:old.importResultId || null, homeGoals:old.homeGoals, awayGoals:old.awayGoals, verifiedAt:old.verifiedAt || null, source:old.source || null, events:old.events || []});
                      const resultRecord = old
                        ? {...old, homeGoals:item.home_goals, awayGoals:item.away_goals, verifiedAt:payload.verified_at, updatedAt:nowISO(), importedAt:nowISO(), importResultId:payload.result_id, source:payload.source, events:item.events || [], revisionNumber:Number(old.revisionNumber || 1) + 1, revisionHistory:history}
                        : {...base(), matchId:match.id, homeGoals:item.home_goals, awayGoals:item.away_goals, verifiedAt:payload.verified_at, importedAt:nowISO(), importResultId:payload.result_id, source:payload.source, events:item.events || [], revisionNumber:1, revisionHistory:[]};
                      tx.objectStore('match_results').put(resultRecord);
                      match.status = 'FINISHED'; match.updatedAt = nowISO();
                      tx.objectStore('matches').put(match);
                      const matchPredictions = currentPredictions.filter(p => p.matchId === match.id && ['CONFIRMED','SETTLED'].includes(p.status));
                      for (const pred of matchPredictions) {
                        const isHome = pred.snapshot?.selectedTeam === 'home';
                        const goalDiff = isHome ? item.home_goals - item.away_goals : item.away_goals - item.home_goals;
                        const settlement = settleHandicap(goalDiff, pred.selectedLine, pred.selectedOdds);
                        const prior = currentPerformance.find(v => v.predictionId === pred.id);
                        const perfHistory = Array.isArray(prior?.revisionHistory) ? prior.revisionHistory.slice() : [];
                        if (prior) perfHistory.push({settlement:prior.settlement, simulatedProfit:prior.simulatedProfit, evaluatedAt:prior.evaluatedAt});
                        const perfRecord = prior
                          ? {...prior, settlement:settlement.settlement, simulatedProfit:settlement.netReturnPerUnit, evaluatedAt:nowISO(), updatedAt:nowISO(), revisionHistory:perfHistory}
                          : {...base(), predictionId:pred.id, settlement:settlement.settlement, simulatedProfit:settlement.netReturnPerUnit, evaluatedAt:nowISO(), revisionHistory:[]};
                        tx.objectStore('performance_records').put(perfRecord);
                        pred.status = 'SETTLED'; pred.updatedAt = nowISO();
                        tx.objectStore('predictions').put(pred);
                      }
                    }
                  } catch (error) {
                    processingError = error;
                    try { tx.abort(); } catch (_) {}
                  }
                };
                requests.forEach(req => {
                  req.onsuccess = () => { ready += 1; if (ready === requests.length) process(); };
                  req.onerror = () => { processingError = req.error || new Error('อ่านข้อมูลในเครื่องไม่สำเร็จ'); try { tx.abort(); } catch (_) {} };
                });
                await completion;
                setMessage('#result-import-message', 'นำเข้าผลสำเร็จ ' + resolved.length + ' คู่ · อัปเดตสถานะหลังแข่งขันและประเมินคำทำนายแล้ว', 'success');
                await render();
              } catch (error) {
                confirmButton.disabled = false;
                setMessage('#result-import-message', error instanceof Error ? error.message : 'ไม่สามารถนำเข้าผลการแข่งขันได้', 'error');
              }
            });
          } catch (error) {
            setMessage('#result-import-message', error instanceof Error ? error.message : 'อ่านหรือเชื่อมโยงไฟล์ผลไม่สำเร็จ', 'error');
          }
        });
    }
