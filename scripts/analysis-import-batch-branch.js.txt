else if (currentView === 'import') {

  const input=document.querySelector('#analysis-import-file'), preview=document.querySelector('#analysis-import-preview');
  const zipReports=async file=>{
    const b=new Uint8Array(await file.arrayBuffer()),v=new DataView(b.buffer),d=new TextDecoder(); let e=-1;
    for(let p=b.length-22;p>=Math.max(0,b.length-65557);p--)if(v.getUint32(p,true)===0x06054b50){e=p;break;}
    if(e<0)throw Error('ไฟล์ ZIP ไม่สมบูรณ์');
    let p=v.getUint32(e+16,true); const count=v.getUint16(e+10,true),out=[];
    for(let i=0;i<count;i++){
      if(v.getUint32(p,true)!==0x02014b50)throw Error('โครงสร้าง ZIP ไม่ถูกต้อง');
      const method=v.getUint16(p+10,true),size=v.getUint32(p+20,true),nl=v.getUint16(p+28,true),el=v.getUint16(p+30,true),cl=v.getUint16(p+32,true),off=v.getUint32(p+42,true),name=d.decode(b.slice(p+46,p+46+nl)); p+=46+nl+el+cl;
      if(!name.toLowerCase().endsWith('.json')||name.startsWith('__MACOSX/'))continue;
      if(v.getUint32(off,true)!==0x04034b50)throw Error('อ่าน ZIP ไม่ได้: '+name);
      const start=off+30+v.getUint16(off+26,true)+v.getUint16(off+28,true),raw=b.slice(start,start+size);
      let txt;
      if(method===0)txt=d.decode(raw);
      else if(method===8){if(!('DecompressionStream' in window))throw Error('เบราว์เซอร์ไม่รองรับ ZIP แบบบีบอัด');txt=await new Response(new Blob([raw]).stream().pipeThrough(new DecompressionStream('deflate-raw'))).text();}
      else throw Error('วิธีบีบอัด ZIP ไม่รองรับ: '+name);
      try{out.push({name,value:JSON.parse(txt)});}catch{throw Error('JSON ภายใน ZIP เสียหาย: '+name);}
    }
    if(!out.length)throw Error('ไม่พบ JSON ใน ZIP'); return out;
  };
  const unpack=(obj,name)=>{
    if(Array.isArray(obj))return obj.map((value,i)=>({name:name+' #'+(i+1),value}));
    if(obj&&Array.isArray(obj.reports)){
      if(obj.schema_version!=='1.0.0'||obj.report_type!=='football-rich-analysis-batch')throw Error('ส่วนหัว Batch ต้องเป็น football-rich-analysis-batch v1.0.0');
      return obj.reports.map((value,i)=>({name:name+' #'+(i+1),value}));
    }
    return [{name,value:obj}];
  };
  input.addEventListener('change',async()=>{
    preview.innerHTML='';const file=input.files?.[0];if(!file)return;
    if(file.size>20*1024*1024){setMessage('#analysis-import-message','ไฟล์ใหญ่เกิน 20 MB','error');return;}
    try{
      const entries=file.name.toLowerCase().endsWith('.zip')?await zipReports(file):unpack(JSON.parse(await file.text()),file.name);
      if(!entries.length||entries.length>100)throw Error('จำนวนรายงานต้องอยู่ระหว่าง 1–100 คู่');
      const items=entries.map((entry,i)=>{
        try{
          const r=entry.value,eg=r?.analysis?.expected_goals;
          if(eg&&Number.isFinite(Number(eg.home))&&Number.isFinite(Number(eg.away))){
            const c=frProbabilityModel(eg,r.market.selected_team,Number(r.market.handicap_line));
            r.analysis.match_outcome_probabilities??=c.match_outcome_probabilities;r.analysis.handicap_settlement_probabilities??=c.handicap_settlement_probabilities;
            r.analysis.cover_probability??=c.cover_probability;r.recommendation.cover_probability??=c.cover_probability;r.analysis.probability_status='estimated';
          }
          validateAnalysisImport(r);return {report:r,name:entry.name};
        }catch(err){throw Error('คู่ที่ '+(i+1)+' ('+entry.name+'): '+(err?.message||'รูปแบบไม่ถูกต้อง'));}
      });
      const ids=items.map(x=>x.report.report_id),keys=items.map(x=>[x.report.match.competition,x.report.match.home_team,x.report.match.away_team].map(s=>String(s).trim().toLowerCase()).join('|'));
      if(new Set(ids).size!==ids.length)throw Error('พบ report_id ซ้ำในไฟล์');
      if(new Set(keys).size!==keys.length)throw Error('พบคู่แข่งขันซ้ำในไฟล์');
      const old=await records(),oldIds=new Set(old.analyses.map(a=>a.importReportId));
      if(ids.some(id=>oldIds.has(id)))throw Error('มี report_id ที่เคยนำเข้าแล้ว กรุณาตรวจสอบไฟล์');
      const updateExistingOnly=document.querySelector('#analysis-import-existing-only')?.checked!==false;
      const normTeam=v=>String(v??'').normalize('NFD').replace(/[\u0300-\u036f]/g,'').toLowerCase().replace(/\s+/g,' ').trim().replace('czechia','czech republic').replace('turkiye','turkey');
      const normComp=v=>String(v??'').toLowerCase().replace(/[—–]/g,'-').replace(/\s+/g,' ').trim();
      const sameCompetition=(a,b)=>{const x=normComp(a),y=normComp(b);return x===y||x.startsWith(y+' ')||y.startsWith(x+' ')||x.includes('uefa nations league')&&y.includes('uefa nations league');};
      const sameKickoff=(a,b)=>{const x=Date.parse(a??''),y=Date.parse(b??'');return Number.isFinite(x)&&Number.isFinite(y)&&Math.abs(x-y)<=120000;};
      const plan=items.map(({report})=>{
        const m=old.matches.filter(x=>sameCompetition(x.league,report.match.competition)&&normTeam(x.homeTeam)===normTeam(report.match.home_team)&&normTeam(x.awayTeam)===normTeam(report.match.away_team)&&(!updateExistingOnly||sameKickoff(x.kickoffAt,report.match.kickoff_at)));
        if(m.length>1)throw Error('พบคู่แข่งขันที่ตรงกันหลายรายการในแอป: '+report.match.home_team+' vs '+report.match.away_team+' · ระบบหยุดเพื่อป้องกันการอัปเดตผิด Match ID');

      });
      const pct=x=>x==null?'—':(Number(x)*100).toFixed(1)+'%';

'+esc(r.match.home_team)+' VS '+esc(r.match.away_team)+'</b><small>'+esc(r.match.competition)+'</small><p class="muted">'+(r.match.kickoff_at?esc(fmtDate(r.match.kickoff_at)):'ไม่ระบุเวลา')+' · '+esc(r.report_id)+'</p>'+(plan[i].match?'<p class="notice">จะอัปเดต Match ID เดิม: '+esc(plan[i].match.id)+'</p>':'<p class="notice">จะสร้างแมตช์ใหม่</p>')+'<div class="metrics"><span>AH <b>'+Number(r.market.handicap_line).toFixed(2)+'</b></span><span>Cover <b>'+pct(r.analysis.cover_probability)+'</b></span></div><p>'+esc(r.recommendation.summary)+'</p></article>').join('')+'<p class="notice">'+(updateExistingOnly?'โหมดปลอดภัย: ทุกคู่เชื่อมกับ Match ID เดิมเท่านั้น และเพิ่มเฉพาะบทวิเคราะห์เวอร์ชันใหม่':'โหมดทั่วไป: อาจสร้างแมตช์ใหม่สำหรับคู่ที่ไม่พบ')+' · ทุกคู่จะถูกบันทึกในธุรกรรมเดียว หากผิดพลาดจะไม่บันทึกบางส่วน</p><button class="primary" id="confirm-analysis-import">นำเข้าทั้งหมด '+items.length+' คู่</button></div>';
